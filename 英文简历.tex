\section{Research Experience}
\vspace{-0.4mm}
{\small\textit{All work conducted at NUS NExT++ Research Centre, supervised by Dr.\ Yonghui Yang and Prof.\ Tat-Seng Chua, NUS Chair Professor, Singapore National Academy of Sciences.}}
\resumeSubHeadingListStart

    \resumeProject
      {LLM Deep Safety Alignment: Auditing Shallow Alignment via Representation-Level Diagnosis}
      {First Author, under-going, targeting ICLR 2027, CCF-A}
      {2025--Present}
      {}
    \resumeItemListStart
        \item {Quantified a \textbf{belief-behavior gap}: internal probes detect harmful intent
            with {>}95\% accuracy at any prefix length, yet model compliance surges from {<}2\%
            to {>}50\% as prefix grows, proving alignment failure is a \emph{generation} problem,
            not a \emph{representation} problem.}
        \item {Proposed \textbf{BEACON}: token-wise consistency losses anchor the model's
            generation behavior to its own internal harmfulness judgment, maintaining refusal
            robustness regardless of prefix length. Cuts adversarial attack success from
            $>$50\% to $\leq$5\% across up to 3,000-token prefixes in four models.}
    \resumeItemListEnd

    \resumeProject
      {S\textsuperscript{3}: Suppressing Method-Specific Shortcuts for Generalizable Deepfake Detection}
      {First Author, Targeting CVPR 2027, CCF-A}
      {2025--2026}
      {}
    \resumeItemListStart
    \item {Identified the root cause of cross-domain failure in deepfake detection:
        trained detectors learn to classify forgery \textit{method} rather than
        forgery \textit{presence} — a shortcut that makes out-of-distribution
        collapse structurally inevitable, regardless of backbone or data scale.}
    \item {Designed two independent remedies: gradient projection (NSP) steers
        training away from method-discriminative feature directions; a plug-in
        inference module (NAE) achieves the same effect without backbone retraining.
        Cross-domain AUC gains exceed 10$\times$ the in-domain cost, validated
        across four backbones and a fully held-out open-world benchmark.}
    \resumeItemListEnd

    \resumeProject

      {2026}
      {}
      \resumeItemListStart
